// Rule-based heuristic generator for rich instant offline calculations
function generateHeuristicAnalysis(text, age) {
const detected = [];
let scoreBonus = 0;
marketingTermsDict.forEach(item => {
if(text.toUpperCase().includes(item.term) || text.includes(item.term)) {
detected.push({
term: item.term,
meaning: item.easy,
difficulty: item.diff === '고급' ? '어려움' : (item.diff === '중급' ? '보통' : '쉬움')
});
}
});
// Default fallback if no known terms found in dictionary
if(detected.length === 0) {
detected.push(
{ term: '타임딜', meaning: '마감 임박 한정 할인', difficulty: '쉬움' },
{ term: 'FOMO', meaning: '놓칠까 봐 두려운 심리 자극', difficulty: '보통' }
);
}
// Age-based persuasion adjust
let posVal = 70, neuVal = 20, negVal = 10;
let pScore = 80;
let hashtags = [];
let insight = "";
let recommended = text;
if(age === '10s' || age === '20s') {
posVal = 78; neuVal = 14; negVal = 8;
pScore = 88;
hashtags = ['#MZ취향저격', '#타임딜_막차', '#소장가치100', '#트렌디쇼핑'];
insight = `${ageTargetLabel(age)} 타겟층은 FOMO, 앰버서더, 팝업스토어 등 최신 SNS 마케팅 어휘에 익숙하며 이를 스타일리시하고 트렌디한 정보로 수용합니다. 용어 사용이 브랜드의 세련된 이미지를 부각시켜 자발적인 SNS 확산을 유도합니다.`;
recommended = text.replace(/ROI/gi, "가성비 대박").replace(/FOMO/gi, "놓치면 후회할").replace(/퍼널/gi, "고객 단계");
} else if(age === '30s' || age === '40s') {
posVal = 65; neuVal = 22; negVal = 13;
pScore = 82;
hashtags = ['#스마트한선택', '#ROI극대화', '#실속혜택', '#직장인필수'];
insight = `${ageTargetLabel(age)} 소비층은 전문적인 마케팅/비즈니스 어휘(ROI, 효율 등)에서 신뢰감을 느끼지만, 불필요한 은어나 유행어에는 피로감을 느낄 수 있습니다. 정량적 수치와 명확한 혜택을 함께 제시하는 것이 효과적입니다.`;
recommended = text.replace(/FOMO 유발/gi, "검증된 혜택의").replace(/마이크로 인플루언서/gi, "전문 유튜버 추천");
} else {
posVal = 48; neuVal = 25; negVal = 27;
pScore = 60;
hashtags = ['#한정특가', '#마감임박', '#추천상품', '#알뜰쇼핑'];
insight = `${ageTargetLabel(age)} 타겟의 경우 영어 약어나 최신 마케터 용어 사용 시 메시지 이해도가 급격히 낮아집니다. 전문 용어를 직관적인 한글 표기('한정 할인', '인기 모델 추천')로 전환해야 구매 전환율이 상승합니다.`;
recommended = "이번 주 한정 수량 할인! 인쇄 및 SNS 인기 모델이 추천하는 맞춤 상품을 놓치지 마세요.";
}
return {
detectedTerms: detected,
sentiment: { positive: posVal, neutral: neuVal, negative: negVal },
persuasionScore: pScore,
persuasionInsight: insight,
emotionalTriggers: ["트렌디함", "전문적 신뢰감", "구매 자극", "약간의 피로도"],
hashtags: hashtags,
metrics: {
compJargon: age === '50s' ? 42 : (age === '10s' ? 75 : 62),
compPlain: 94,
credJargon: 85,
credPlain: 68,
buzzJargon: 82,
buzzPlain: 70,
convJargon: age === '50s' ? 45 : 70,
convPlain: 85
},
recommendedCopy: recommended
};
}
